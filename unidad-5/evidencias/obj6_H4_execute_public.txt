SELECT grantee, routine_name FROM information_schema.role_routine_grants
WHERE routine_schema = 'public' AND routine_name IN ('sp_crear_pedido','fn_autenticar','fn_resetear_password') AND grantee = 'PUBLIC';

SELECT p.oid::regprocedure AS rutina, p.proowner::regrole AS dueno, p.proacl,
       has_function_privilege('app_web', p.oid, 'EXECUTE') AS app_web_puede_ejecutar
FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace
WHERE n.nspname = 'public'
  AND NOT EXISTS (SELECT 1 FROM pg_depend d WHERE d.objid = p.oid AND d.deptype = 'e')
ORDER BY 1;
